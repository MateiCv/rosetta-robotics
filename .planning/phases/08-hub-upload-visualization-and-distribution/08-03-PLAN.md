Create `Cross.toml` at the workspace root for cross-compilation of the Linux aarch64 target. The `cross` tool uses Docker containers for cross-compilation, but the default images lack FFmpeg and HDF5 dev libraries needed by forge.

```toml
[target.aarch64-unknown-linux-gnu]
# Use pre-build hook to install FFmpeg and HDF5 dev libs in the cross container.
# cross runs this script inside the Docker container before building.
pre-build = [
    "dpkg --add-architecture arm64",
    "apt-get update",
    "apt-get install -y libavcodec-dev:arm64 libavformat-dev:arm64 libavutil-dev:arm64 libavfilter-dev:arm64 libavdevice-dev:arm64 libswscale-dev:arm64 libswresample-dev:arm64 libhdf5-dev:arm64 pkg-config"
]
```

**Note:** If `pre-build` with architecture-specific packages does not work cleanly in cross's Docker setup, fall back to using a custom Dockerfile:
1. Create `.cross/Dockerfile.aarch64` based on `ghcr.io/cross-rs/aarch64-unknown-linux-gnu:main`
2. Add the FFmpeg/HDF5 dev package installation in the Dockerfile
3. Reference it in Cross.toml: `image = "forge-cross-aarch64"` with `dockerfile = ".cross/Dockerfile.aarch64"`

If cross-compilation for aarch64 proves too complex due to system library dependencies, make it a best-effort target: build it in the workflow but allow it to fail with `continue-on-error: true`, and document it as requiring native compilation.

Create `.github/workflows/release.yml`:

```yaml
name: Release

on:
  push:
    tags: ['v*']

permissions:
  contents: write

env:
  CARGO_TERM_COLOR: always

jobs:
  build:
    name: Build (${{ matrix.name }})
    strategy:
      fail-fast: false
      matrix:
        include:
          - target: x86_64-unknown-linux-gnu
            os: ubuntu-22.04
            name: linux-x86_64
            use_cross: false
          - target: aarch64-unknown-linux-gnu
            os: ubuntu-22.04
            name: linux-aarch64
            use_cross: true
          - target: x86_64-apple-darwin
            os: macos-13
            name: macos-x86_64
            use_cross: false
          - target: aarch64-apple-darwin
            os: macos-14
            name: macos-aarch64
            use_cross: false
          - target: x86_64-pc-windows-msvc
            os: windows-latest
            name: windows-x86_64
            use_cross: false
    runs-on: ${{ matrix.os }}
    steps:
      - uses: actions/checkout@v4

      - name: Install Rust toolchain
        uses: actions-rust-lang/setup-rust-toolchain@v1
        with:
          toolchain: stable
          target: ${{ matrix.target }}

      - name: Install FFmpeg (Linux native)
        if: runner.os == 'Linux' && !matrix.use_cross
        run: |
          sudo apt-get update
          sudo apt-get install -y \
            libavcodec-dev libavformat-dev libavutil-dev \
            libavfilter-dev libavdevice-dev libswscale-dev \
            libswresample-dev pkg-config

      - name: Install HDF5 (Linux native)
        if: runner.os == 'Linux' && !matrix.use_cross
        run: sudo apt-get install -y libhdf5-dev

      - name: Install FFmpeg (macOS)
        if: runner.os == 'macOS'
        run: brew install ffmpeg pkg-config

      - name: Install HDF5 (macOS)
        if: runner.os == 'macOS'
        run: brew install hdf5

      - name: Install FFmpeg (Windows)
        if: runner.os == 'Windows'
        run: |
          choco install ffmpeg-full -y
          choco install pkgconfiglite -y

      - name: Install cross (Linux aarch64)
        if: matrix.use_cross
        run: cargo install cross --locked

      - name: Build (cross)
        if: matrix.use_cross
        run: cross build --release --target ${{ matrix.target }} -p forge-cli

      - name: Build (cargo)
        if: "!matrix.use_cross"
        run: cargo build --release --target ${{ matrix.target }} -p forge-cli

      - name: Package (Unix)
        if: runner.os != 'Windows'
        run: |
          cd target/${{ matrix.target }}/release
          tar czf ../../../forge-${{ github.ref_name }}-${{ matrix.name }}.tar.gz forge
          cd ../../..

      - name: Package (Windows)
        if: runner.os == 'Windows'
        shell: pwsh
        run: |
          Compress-Archive -Path target/${{ matrix.target }}/release/forge.exe -DestinationPath forge-${{ env.TAG }}-${{ matrix.name }}.zip
        env:
          TAG: ${{ github.ref_name }}

      - name: Upload artifact
        uses: actions/upload-artifact@v4
        with:
          name: forge-${{ matrix.name }}
          path: |
            forge-*.tar.gz
            forge-*.zip

  release:
    name: Create Release
    needs: build
    runs-on: ubuntu-latest
    steps:
      - uses: actions/download-artifact@v4
        with:
          merge-multiple: true

      - name: Create GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          generate_release_notes: true
          files: |
            forge-*.tar.gz
            forge-*.zip
```

Key design decisions:
- `fail-fast: false` so one platform failure does not block others
- `cross` only used for Linux aarch64 (the only target that needs cross-compilation); all others build natively on their respective runners
- macOS x86_64 uses macos-13 (Intel runner), macOS aarch64 uses macos-14 (Apple Silicon runner)
- Windows uses PowerShell Compress-Archive for .zip; Unix uses tar for .tar.gz
- FFmpeg/HDF5 installation mirrors the existing ci.yml patterns
- `generate_release_notes: true` auto-generates changelog from commits since last tag
- The binary name is `forge` (from `[[bin]] name = "forge"` in forge-cli's Cargo.toml)
