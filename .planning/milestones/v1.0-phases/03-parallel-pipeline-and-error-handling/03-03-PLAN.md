```rust
use indicatif::{ProgressBar, ProgressStyle, ProgressState, ProgressTracker};
use sysinfo::System;
use std::fmt::Write;

/// Custom progress tracker that displays current memory usage.
struct MemoryTracker;

impl ProgressTracker for MemoryTracker {
    fn clone_box(&self) -> Box<dyn ProgressTracker> {
        Box::new(MemoryTracker)
    }
    fn tick(&mut self, _: &ProgressState, _: std::time::Instant) {}
    fn reset(&mut self, _: &ProgressState, _: std::time::Instant) {}
    fn write(&self, _: &ProgressState, w: &mut dyn Write) {
        // NOTE: Do NOT refresh sysinfo on every call -- cache with a timer.
        // For simplicity, refresh here (called ~1/sec by indicatif tick).
        let mut sys = System::new();
        sys.refresh_memory();
        let used_gb = sys.used_memory() as f64 / 1_073_741_824.0;
        write!(w, "{used_gb:.1}GB").unwrap();
    }
}

/// Progress reporting modes for conversion.
pub enum ProgressMode {
    /// Single-line progress bar with ETA, throughput, memory (default).
    Bar,
    /// No progress output (--quiet).
    Quiet,
    /// JSON summary at end (--json).
    Json,
    /// Per-episode verbose log lines (--verbose).
    Verbose,
}

/// Wraps indicatif ProgressBar with memory tracking and mode awareness.
pub struct ProgressReporter {
    bar: Option<ProgressBar>,
    mode: ProgressMode,
}

impl ProgressReporter {
    pub fn new(total_episodes: u64, mode: ProgressMode) -> Self {
        let bar = match &mode {
            ProgressMode::Bar => {
                let pb = ProgressBar::new(total_episodes);
                let style = ProgressStyle::with_template(
                    "{spinner:.green} [{bar:40.cyan/blue}] {pos}/{len} episodes | \
                     {per_sec} | ETA: {eta} | mem: {mem}"
                ).unwrap()
                .with_key("mem", MemoryTracker)
                .progress_chars("=>-");
                pb.set_style(style);
                pb.enable_steady_tick(std::time::Duration::from_millis(200));
                Some(pb)
            }
            ProgressMode::Verbose => {
                // Use a hidden progress bar for timing, but log each episode
                let pb = ProgressBar::hidden();
                Some(pb)
            }
            _ => None,
        };
        Self { bar, mode }
    }

    /// Increment progress by one episode. For verbose mode, also log the episode.
    pub fn inc(&self, episode_index: usize) {
        if let Some(ref pb) = self.bar {
            pb.inc(1);
        }
        if matches!(self.mode, ProgressMode::Verbose) {
            eprintln!("  episode {episode_index}: done");
        }
    }

    /// Mark a failed episode in verbose mode.
    pub fn fail(&self, episode_index: usize, error: &str) {
        if matches!(self.mode, ProgressMode::Verbose) {
            eprintln!("  episode {episode_index}: FAILED - {error}");
        }
    }

    /// Finish the progress bar.
    pub fn finish(&self) {
        if let Some(ref pb) = self.bar {
            pb.finish_with_message("done");
        }
    }
}

/// Determine ProgressMode from ConversionConfig.
impl From<&forge_core::ConversionConfig> for ProgressMode {
    fn from(config: &forge_core::ConversionConfig) -> Self {
        if config.quiet {
            ProgressMode::Quiet
        } else if config.json_output {
            ProgressMode::Json
        } else if config.verbose {
            ProgressMode::Verbose
        } else {
            ProgressMode::Bar
        }
    }
}
```

**2. Create `crates/forge-pipeline/src/prefetch.rs`:**

Implement background episode prefetch using crossbeam bounded channel:

```rust
use crossbeam_channel::{bounded, Receiver};
use forge_core::{Episode, ForgeResult};

/// Prefetch episodes from a reader into a bounded channel.
///
/// Spawns a background thread that reads episodes from the iterator and
/// sends them into a bounded channel. The consumer (Converter) receives
/// episodes from the channel. When the channel is full (capacity reached),
/// the producer blocks until the consumer reads -- natural backpressure.
///
/// This overlaps episode reading/decoding with episode writing, improving
/// throughput for I/O-bound workloads.
pub fn prefetch_episodes(
    episodes: Box<dyn Iterator<Item = ForgeResult<Episode>>>,
    buffer_size: usize,
) -> Receiver<ForgeResult<Episode>> {
    let (tx, rx) = bounded(buffer_size);

    std::thread::spawn(move || {
        for episode in episodes {
            if tx.send(episode).is_err() {
                break; // Consumer dropped the receiver
            }
        }
        // tx drops here, closing the channel
    });

    rx
}
```

The buffer_size should default to 2 (decode 2 episodes ahead). This is small enough to avoid memory pressure but large enough to hide I/O latency.

    let marker_path = output_path.join(format!("episode_{:06}.failed", episode_index));

```rust
fn run_convert(
    source: &str,
    output: &str,
    target_format: &str,
    fail_on_error: bool,
    deterministic: bool,
    threads: Option<usize>,
    quiet: bool,
    json: bool,
    verbose: bool,
) {
    let config = ConversionConfig {
        target_format: Some(target_format.to_string()),
        fail_on_error,
        deterministic,
        num_workers: threads.unwrap_or(0),
        quiet,
        json_output: json,
        verbose,
        ..ConversionConfig::default()
    };

    let converter = Converter::new(config);

    // Only print header if not quiet and not json
    if !quiet && !json {
        eprintln!("Converting: {} -> {}", source, output);
        eprintln!("Target format: {}", target_format);
    }

    match converter.convert(Path::new(source), Path::new(output)) {
        Ok(result) => {
            // Print summary (unless json mode -- Converter already printed JSON)
            if !quiet && !json {
                eprintln!("\nConversion complete:");
                eprintln!("  Episodes converted: {}", result.episodes_converted);
                eprintln!("  Episodes failed:    {}", result.episodes_failed);
                eprintln!("  Total frames:       {}", result.total_frames);
            }

            // Non-zero exit code if any episodes failed
            if result.episodes_failed > 0 {
                std::process::exit(1);
            }
        }
        Err(e) => {
            eprintln!("Error: {e}");
            std::process::exit(1);
        }
    }
}
```
