   pub fn detect_format_scored(path: &Path) -> ForgeResult<Vec<DetectionResult>> {

           Box::new(lerobot_v3::LeRobotV3Detector),  // v3 before v2
           Box::new(lerobot_v2::LeRobotV2Detector),  // v2 after v3
           Box::new(robodm::RoboDMDetector),          // .vla extension unique
           Box::new(rlds::RLDSDetector),              // .tfrecord files
           Box::new(zarr::ZarrDetector),              // .zarr / .zarray

       ];
       // ... collect, sort, warn on ambiguity (already implemented in 05-01)
   }
