   ```rust
   #[derive(Debug, Clone)]
   pub struct ReadOutput<T> {
       pub data: T,
       pub warnings: Vec<ReadWarning>,
   }

   impl<T> ReadOutput<T> {
       pub fn new(data: T) -> Self {
           Self { data, warnings: Vec::new() }
       }

       pub fn with_warnings(data: T, warnings: Vec<ReadWarning>) -> Self {
           Self { data, warnings }
       }

       /// Returns true if any warning has Error severity.
       pub fn has_errors(&self) -> bool {
           self.warnings.iter().any(|w| w.severity == WarningSeverity::Error)
       }

       /// Returns a human-readable summary line, e.g. "47/50 episodes read, 3 skipped (truncated data)".
       pub fn summary(&self) -> String
       where T: AsRef<[Episode]> {
           // Implementation: count data length, count skipped from warnings, format summary string.
           // Caller can also build custom summaries from the warnings vec.
           let total_read = self.data.as_ref().len();
           let skipped = self.warnings.iter().filter(|w| w.severity == WarningSeverity::Error).count();
           if skipped > 0 {
               format!("{}/{} episodes read, {} skipped", total_read, total_read + skipped, skipped)
           } else {
               format!("{} episodes read", total_read)
           }
       }
   }
   ```
   This is the return path for warnings per the locked user decision: "collect all warnings and show a summary after reading completes."
